inline constexpr std::uint64_t kFnvOffsetBasis = 0xcbf29ce484222325ULL;
inline constexpr std::uint64_t kFnvPrime = 0x100000001b3ULL;

constexpr std::uint64_t Fnv1a(std::string_view s) noexcept {
  std::uint64_t h = kFnvOffsetBasis;
  for (const char c : s) {
    h ^= static_cast<std::uint64_t>(static_cast<unsigned char>(c));
    h *= kFnvPrime;
  }
  return h;
}

static_assert(Fnv1a("") == kFnvOffsetBasis, "empty string hashes to the offset basis");

// A ring buffer with a compile-time capacity. Overwrites the oldest element when full, which is
// what you want for a telemetry tail and emphatically not what you want for a work queue.
template <typename T, std::size_t N>
class RingBuffer {
  static_assert(N > 0, "RingBuffer needs a non-zero capacity");

 public:
  using value_type = T;
  using size_type = std::size_t;

  constexpr size_type capacity() const noexcept { return N; }
  constexpr size_type size() const noexcept { return size_; }
  constexpr bool empty() const noexcept { return size_ == 0; }
  constexpr bool full() const noexcept { return size_ == N; }

  void push(T value) {
    slots_[tail_] = std::move(value);
    tail_ = (tail_ + 1) % N;
    if (full()) {
      head_ = (head_ + 1) % N;  // drop the oldest
    } else {
      ++size_;
    }
  }

  std::optional<T> pop() {
    if (empty()) return std::nullopt;
    T out = std::move(slots_[head_]);
    head_ = (head_ + 1) % N;
    --size_;
    return out;
  }

  const T& operator[](size_type i) const {
    if (i >= size_) throw std::out_of_range("RingBuffer index " + std::to_string(i));
    return slots_[(head_ + i) % N];
  }

  template <typename Fn>
  void for_each(Fn&& fn) const {
    for (size_type i = 0; i < size_; ++i) fn((*this)[i]);
  }

 private:
  std::array<T, N> slots_{};
  size_type head_ = 0;
  size_type tail_ = 0;
  size_type size_ = 0;
};

struct Interval {
  std::int64_t lo = 0;
  std::int64_t hi = 0;  // half-open: [lo, hi)

  constexpr bool valid() const noexcept { return lo < hi; }
  constexpr bool overlaps(const Interval& other) const noexcept {
    return lo < other.hi && other.lo < hi;
  }
  friend constexpr bool operator<(const Interval& a, const Interval& b) noexcept {
    return a.lo != b.lo ? a.lo < b.lo : a.hi < b.hi;
  }
  friend constexpr bool operator==(const Interval& a, const Interval& b) noexcept {
    return a.lo == b.lo && a.hi == b.hi;
  }
};

// Sorted, coalescing interval set. O(log n) lookup, O(n) worst-case insert -- fine for the few
// hundred intervals this was written for, and a red-black tree would have been a lie about the
// problem size.
class IntervalSet {
 public:
  void insert(Interval iv) {
    if (!iv.valid()) return;
    auto it = std::lower_bound(items_.begin(), items_.end(), iv);
    if (it != items_.begin() && std::prev(it)->hi >= iv.lo) --it;
    auto first = it;
    while (it != items_.end() && it->lo <= iv.hi) {
      iv.lo = std::min(iv.lo, it->lo);
      iv.hi = std::max(iv.hi, it->hi);
      ++it;
    }
    it = items_.erase(first, it);
    items_.insert(it, iv);
  }

  bool contains(std::int64_t point) const {
    auto it = std::upper_bound(items_.begin(), items_.end(), Interval{point, point + 1});
    if (it == items_.begin()) return false;
    --it;
    return it->lo <= point && point < it->hi;
  }

  std::int64_t measure() const {
    std::int64_t total = 0;
    for (const auto& iv : items_) total += iv.hi - iv.lo;
    return total;
  }

  const std::vector<Interval>& items() const noexcept { return items_; }

 private:
  std::vector<Interval> items_;
};

class ScopedTimer {
 public:
  explicit ScopedTimer(double* sink) : sink_(sink), start_(Now()) {}
  ScopedTimer(const ScopedTimer&) = delete;
  ScopedTimer& operator=(const ScopedTimer&) = delete;
  ~ScopedTimer() {
    if (sink_ != nullptr) *sink_ += Now() - start_;
  }

 private:
  static double Now() noexcept;
  double* sink_ = nullptr;
  double start_ = 0.0;
};

}  // namespace toybox

#ifdef TOYBOX_SELF_TEST
#include <cstdio>
int main() {
  toybox::RingBuffer<int, 4> rb;
  for (int i = 0; i < 7; ++i) rb.push(i);
  assert(rb.size() == 4 && rb[0] == 3 && rb[3] == 6);

  toybox::IntervalSet set;
  set.insert({0, 10});
  set.insert({20, 30});
  set.insert({8, 22});  // coalesces all three
  assert(set.items().size() == 1u);
  assert(set.measure() == 30);
  assert(set.contains(15) && !set.contains(30));
  std::printf("ok: 0x%016llx\n", static_cast<unsigned long long>(toybox::Fnv1a("toybox")));
  return 0;
}
#endif  // TOYBOX_SELF_TEST
