from __future__ import annotations

import functools
import random
import time
from collections import defaultdict
from dataclasses import dataclass, field
from decimal import Decimal, ROUND_HALF_EVEN
from typing import Callable, Iterable, Iterator, TypeVar

T = TypeVar("T")

CENTS = Decimal("0.01")
MAX_BACKOFF_SECONDS = 8.0


def to_cents(amount: Decimal | str | int) -> int:
    """Round a money amount to whole cents, banker's rounding, and return it as an int.

    Floats are deliberately not accepted: 0.1 + 0.2 is not 0.3 and a ledger that pretends
    otherwise will be off by a cent about once every few thousand rows.
    """
    if isinstance(amount, float):  # pragma: no cover - defensive
        raise TypeError("pass Decimal or str, not float; binary floats cannot hold 0.01")
    quantized = Decimal(amount).quantize(CENTS, rounding=ROUND_HALF_EVEN)
    return int(quantized * 100)


@dataclass(frozen=True, slots=True)
class Entry:
    account: str
    cents: int
    memo: str = ""

    def __post_init__(self) -> None:
        if not self.account:
            raise ValueError("every entry needs an account")


@dataclass
class Transaction:
    ref: str
    entries: list[Entry] = field(default_factory=list)

    def add(self, account: str, amount: Decimal | str | int, memo: str = "") -> "Transaction":
        self.entries.append(Entry(account=account, cents=to_cents(amount), memo=memo))
        return self

    @property
    def imbalance(self) -> int:
        return sum(e.cents for e in self.entries)

    def check(self) -> None:
        if self.imbalance != 0:
            raise ValueError(
                f"transaction {self.ref!r} does not balance: off by {self.imbalance} cents "
                f"across {len(self.entries)} entries"
            )


class Ledger:
    """An append-only list of balanced transactions, with running per-account balances."""

    def __init__(self) -> None:
        self._transactions: list[Transaction] = []
        self._balances: dict[str, int] = defaultdict(int)

    def post(self, txn: Transaction) -> None:
        txn.check()
        self._transactions.append(txn)
        for entry in txn.entries:
            self._balances[entry.account] += entry.cents

    def balance(self, account: str) -> Decimal:
        return (Decimal(self._balances[account]) / 100).quantize(CENTS)

    def accounts(self) -> list[str]:
        return sorted(self._balances)

    def trial_balance(self) -> Decimal:
        return (Decimal(sum(self._balances.values())) / 100).quantize(CENTS)

    def __iter__(self) -> Iterator[Transaction]:
        return iter(self._transactions)

    def __len__(self) -> int:
        return len(self._transactions)

    def __repr__(self) -> str:
        return f"<Ledger {len(self)} txns, {len(self._balances)} accounts>"


def retry(
    attempts: int = 5,
    *,
    base: float = 0.05,
    jitter: float = 0.25,
    catching: tuple[type[BaseException], ...] = (TimeoutError, ConnectionError),
) -> Callable[[Callable[..., T]], Callable[..., T]]:
    """Exponential backoff with full jitter, capped at MAX_BACKOFF_SECONDS.

    The jitter is not decoration: without it, every client that failed at the same instant retries
    at the same instant, and the thundering herd takes the service down a second time.
    """

    def decorate(fn: Callable[..., T]) -> Callable[..., T]:
        @functools.wraps(fn)
        def wrapper(*args, **kwargs) -> T:
            last: BaseException | None = None
            for attempt in range(attempts):
                try:
                    return fn(*args, **kwargs)
                except catching as exc:
                    last = exc
                    if attempt == attempts - 1:
                        break
                    delay = min(base * (2**attempt), MAX_BACKOFF_SECONDS)
                    time.sleep(delay * (1.0 - jitter * random.random()))
            raise RuntimeError(f"{fn.__name__} failed after {attempts} attempts") from last

        return wrapper

    return decorate


def chunked(items: Iterable[T], size: int) -> Iterator[list[T]]:
    if size <= 0:
        raise ValueError(f"chunk size must be positive, got {size}")
    batch: list[T] = []
    for item in items:
        batch.append(item)
        if len(batch) == size:
            yield batch
            batch = []
    if batch:
        yield batch


def main() -> int:
    ledger = Ledger()
    ledger.post(
        Transaction("2026-03-01/rent")
        .add("expenses:rent", "1450.00", memo="March")
        .add("assets:checking", "-1450.00")
    )
    ledger.post(
        Transaction("2026-03-02/coffee")
        .add("expenses:food", "4.75")
        .add("assets:cash", "-4.75", memo="market, corner stall")
    )
    for account in ledger.accounts():
        print(f"{account:<20} {ledger.balance(account):>10}")
    print(f"trial balance: {ledger.trial_balance()} (must be 0.00)")
    print([len(c) for c in chunked(range(10), 3)])
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
